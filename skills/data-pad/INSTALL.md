```bash
# 1. Clone the project (skip if you already have it)
git clone https://github.com/<your-github-org>/data-pad.git ~/projects/data-pad

# 2. Install dependencies
cd ~/projects/data-pad && npm install

# 3. (Optional) Pin the location with an env var
export DATA_PAD_HOME=~/projects/data-pad
# Add the above to ~/.bashrc or ~/.zshrc to make it permanent.
```
